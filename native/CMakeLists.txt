# English is source text and requires no Linguist dependency. Community catalogs
# are compiled and embedded when present, so every package uses the same data.
function(ranslator_find_lrelease)
  if(DEFINED ENV{LRELEASE} AND NOT "$ENV{LRELEASE}" STREQUAL "")
    set(RANSLATOR_LRELEASE "$ENV{LRELEASE}" CACHE FILEPATH "Qt 6 lrelease executable" FORCE)
  endif()
  find_program(RANSLATOR_LRELEASE NAMES lrelease6 lrelease-qt6 lrelease
    HINTS "${QT_HOST_PATH}/bin" "${QT6_INSTALL_PREFIX}/bin" /usr/lib/qt6/bin
          /opt/homebrew/opt/qt/bin /usr/local/opt/qt/bin REQUIRED)
  execute_process(COMMAND "${RANSLATOR_LRELEASE}" -version
    RESULT_VARIABLE status OUTPUT_VARIABLE version ERROR_VARIABLE error)
  if(NOT status EQUAL 0 OR NOT "${version}${error}" MATCHES "version 6\\.")
    message(FATAL_ERROR "UI catalogs require Qt 6 lrelease. Set LRELEASE or RANSLATOR_LRELEASE to its executable path.")
  endif()
endfunction()
file(GLOB RANSLATOR_CATALOGS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../translations/ranslator_*.ts")
list(FILTER RANSLATOR_CATALOGS EXCLUDE REGEX "/ranslator_en\\.ts$")
if(RANSLATOR_CATALOGS)
  ranslator_find_lrelease()
  set(RANSLATOR_QRC "<RCC><qresource prefix=\"/i18n\">\n")
  set(RANSLATOR_QM_FILES)
  foreach(catalog IN LISTS RANSLATOR_CATALOGS)
    get_filename_component(name "${catalog}" NAME_WE)
    set(qm "${CMAKE_CURRENT_BINARY_DIR}/translations/${name}.qm")
    add_custom_command(OUTPUT "${qm}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/translations"
      COMMAND "${RANSLATOR_LRELEASE}" -nounfinished "${catalog}" -qm "${qm}"
      DEPENDS "${catalog}" VERBATIM)
    list(APPEND RANSLATOR_QM_FILES "${qm}")
    string(APPEND RANSLATOR_QRC "<file alias=\"${name}.qm\">${qm}</file>\n")
  endforeach()
  string(APPEND RANSLATOR_QRC "</qresource></RCC>\n")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ranslator_translations.qrc" "${RANSLATOR_QRC}")
  qt_add_resources(RANSLATOR_I18N_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/ranslator_translations.qrc")
  target_sources(ranslator_ui PRIVATE ${RANSLATOR_I18N_RESOURCES})
  target_compile_definitions(ranslator_ui PRIVATE RANSLATOR_BUNDLED_TRANSLATIONS)
  if(TARGET ui_tests)
    target_sources(ui_tests PRIVATE ${RANSLATOR_I18N_RESOURCES})
    target_compile_definitions(ui_tests PRIVATE RANSLATOR_BUNDLED_TRANSLATIONS)
  endif()
endif()
